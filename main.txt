import tkinter as tk
from tkinter import messagebox
import re
import threading
import urllib.request
import ast
import operator
import math

class AICalculatorPro:
    def __init__(self, root):
        self.root = root
        self.version = "2.0.0"  # تحديث كبرى لدعم العمليات المتقدمة
        self.root.title(f"AICalculatorPro v{self.version}")
        self.root.geometry("380x600")
        
        # --- تفعيل التكبير والتصغير بالكامل ---
        self.root.resizable(True, True)
        self.root.minsize(320, 500)
        # -------------------------------------
        
        self.root.configure(bg="#1e1e2e")

        self.expression = ""
        self.display_var = tk.StringVar(value="0")
        
        display_frame = tk.Frame(self.root, bg="#1e1e2e", bd=10)
        display_frame.pack(expand=False, fill="both")

        self.display_label = tk.Label(
            display_frame,
            textvariable=self.display_var,
            font=("Consolas", 24, "bold"),
            anchor="e",
            bg="#181825",
            fg="#cdd6f4",
            padx=15,
            pady=15
        )
        self.display_label.pack(expand=True, fill="both")

        buttons_frame = tk.Frame(self.root, bg="#1e1e2e")
        buttons_frame.pack(expand=True, fill="both", padx=10, pady=10)

        # تصميم الأزرار المحدث (يشمل العمليات المتقدمة)
        buttons = [
            ('C', 0, 0, "#f38ba8"), ('(', 0, 1, "#89b4fa"), (')', 0, 2, "#89b4fa"), ('/', 0, 3, "#fab387"),
            ('sin', 1, 0, "#89b4fa"), ('cos', 1, 1, "#89b4fa"), ('tan', 1, 2, "#89b4fa"), ('*', 1, 3, "#fab387"),
            ('x^2', 2, 0, "#89b4fa"), ('sqrt', 2, 1, "#89b4fa"), ('pi', 2, 2, "#fab387"), ('-', 2, 3, "#fab387"),
            ('7', 3, 0, "#313244"), ('8', 3, 1, "#313244"), ('9', 3, 2, "#313244"), ('+', 3, 3, "#fab387"),
            ('4', 4, 0, "#313244"), ('5', 4, 1, "#313244"), ('6', 4, 2, "#313244"), ('DEL', 4, 3, "#f38ba8"),
            ('1', 5, 0, "#313244"), ('2', 5, 1, "#313244"), ('3', 5, 2, "#313244"), ('=', 5, 3, "#a6e3a1"),
            ('0', 6, 0, "#313244"), ('.', 6, 1, "#313244"), ('e', 6, 2, "#fab387")
        ]

        for i in range(7):
            buttons_frame.grid_rowconfigure(i, weight=1)
        for j in range(4):
            buttons_frame.grid_columnconfigure(j, weight=1)

        for text, row, col, color in buttons:
            # جعل زر الصفر يمتد على عمودين إن أردت أو يظل في مكانه (هنا يأخذ خانة واحدة مع بقية الأزرار بمرونة)
            btn = tk.Button(
                buttons_frame,
                text=text,
                font=("Arial", 12, "bold"),
                bg=color,
                fg="#11111b" if color in ["#f38ba8", "#89b4fa", "#fab387", "#a6e3a1"] else "#cdd6f4",
                activebackground="#45475a",
                activeforeground="#ffffff",
                relief="flat",
                bd=0,
                command=lambda t=text: self.on_button_click(t)
            )
            # دمج زر الصفر أو توزيعه بشكل صحيح
            if text == '0':
                btn.grid(row=row, column=col, columnspan=2, padx=4, pady=4, sticky="nsew")
            elif text == '.':
                btn.grid(row=row, column=col+1, padx=4, pady=4, sticky="nsew")
            elif text == 'e':
                btn.grid(row=row, column=col+1, padx=4, pady=4, sticky="nsew")
            elif text in ['.', 'e'] and col == 1: # لتلافي تداخل الأعمدة بعد الصفر
                pass
            else:
                if text == '.':
                    btn.grid(row=row, column=2, padx=4, pady=4, sticky="nsew")
                elif text == 'e':
                    btn.grid(row=row, column=3, padx=4, pady=4, sticky="nsew")
                else:
                    btn.grid(row=row, column=col, padx=4, pady=4, sticky="nsew")

        # إعادة ضبط توزيع شبكة الأزرار بشكل دقيق لتفادي التداخل
        self.rebuild_grid(buttons_frame, buttons)

        # ربط لوحة المفاتيح
        self.root.bind("<Key>", self.on_key_press)
        self.check_for_updates_silently()

    def rebuild_grid(self, frame, buttons):
        """إعادة ترتيب الأزرار بوضوح ودقة تامة"""
        for widget in frame.winfo_children():
            widget.grid_forget()

        for text, row, col, color in buttons:
            btn = tk.Button(
                frame,
                text=text,
                font=("Arial", 11, "bold"),
                bg=color,
                fg="#11111b" if color in ["#f38ba8", "#89b4fa", "#fab387", "#a6e3a1"] else "#cdd6f4",
                activebackground="#45475a",
                activeforeground="#ffffff",
                relief="flat",
                bd=0,
                command=lambda t=text: self.on_button_click(t)
            )
            if text == '0':
                btn.grid(row=row, column=col, columnspan=2, padx=3, pady=3, sticky="nsew")
            elif text == '.':
                btn.grid(row=row, column=2, padx=3, pady=3, sticky="nsew")
            elif text == 'e':
                btn.grid(row=row, column=3, padx=3, pady=3, sticky="nsew")
            else:
                btn.grid(row=row, column=col, padx=3, pady=3, sticky="nsew")

    def check_for_updates_silently(self):
        def background_task():
            try:
                url = "https://raw.githubusercontent.com/816400/calculator-app/main/version.txt"
                req = urllib.request.Request(url, headers={'User-Agent': 'AICalculatorPro-SecureClient'})
                with urllib.request.urlopen(req, timeout=3) as response:
                    latest_version = response.read().decode('utf-8').strip()
                    if re.match(r'^\d+(\.\d+)*$', latest_version) and latest_version > self.version:
                        self.root.after(0, lambda: messagebox.showinfo(
                            "تحديث متوفر", 
                            f"يتوفر إصدار جديد وآمن ({latest_version})!\nيمكنك تحديث نسختك من مستودع GitHub."
                        ))
            except Exception:
                pass
        threading.Thread(target=background_task, daemon=True).start()

    def on_key_press(self, event):
        char = event.char
        if char in "0123456789+-*/().":
            self.on_button_click(char)
        elif event.keysym in ("Return", "KP_Enter"):
            self.on_button_click("=")
        elif event.keysym == "BackSpace":
            self.on_button_click("DEL")
        elif event.keysym in ("Escape", "c", "C"):
            self.on_button_click("C")

    def safe_eval(self, expr):
        """محلل حسابي مطور وآمن يدعم العمليات الرياضية والمثلثية المتقدمة"""
        operators = {
            ast.Add: operator.add,
            ast.Sub: operator.sub,
            ast.Mult: operator.mul,
            ast.Div: operator.truediv,
            ast.Pow: operator.pow,
            ast.UAdd: operator.pos,
            ast.USub: operator.neg,
        }

        # دوال الثوابت والمثلثات المدعومة بأمان
        safe_names = {
            'pi': math.pi,
            'e': math.e,
            'sin': lambda x: math.sin(math.radians(x)),
            'cos': lambda x: math.cos(math.radians(x)),
            'tan': lambda x: math.tan(math.radians(x)),
            'sqrt': math.sqrt,
        }

        def _eval(node):
            if isinstance(node, ast.Constant):
                if isinstance(node.value, (int, float)):
                    return node.value
                raise TypeError("نوع بيانات غير مدعوم")
            elif isinstance(node, ast.Name):
                if node.id in safe_names:
                    return safe_names[node.id]
                raise NameError(f"متغير غير مسموح به: {node.id}")
            elif isinstance(node, ast.BinOp):
                if type(node.op) in operators:
                    return operators[type(node.op)](_eval(node.left), _eval(node.right))
                raise TypeError("عملية غير مسموح بها")
            elif isinstance(node, ast.UnaryOp):
                if type(node.op) in operators:
                    return operators[type(node.op)](_eval(node.operand))
                raise TypeError("عملية أحادية غير مسموح بها")
            elif isinstance(node, ast.Call):
                if isinstance(node.func, ast.Name) and node.func.id in safe_names:
                    func = safe_names[node.func.id]
                    args = [_eval(arg) for arg in node.args]
                    return func(*args)
                raise TypeError("دالة غير مسموح بها")
            else:
                raise TypeError("تعبير غير صالح")

        parsed_node = ast.parse(expr, mode='eval')
        return _eval(parsed_node.body)

    def on_button_click(self, char):
        if char == "C":
            self.expression = ""
            self.display_var.set("0")
        elif char == "DEL":
            self.expression = self.expression[:-1]
            self.display_var.set(self.expression if self.expression else "0")
        elif char == "=":
            try:
                if len(self.expression) > 150:
                    raise ValueError("التعبير طويل جداً")
                
                # استبدال رموز الواجهة بصيغة يفهمها محلل بايثون (مثل x^2 إلى **2)
                eval_expr = self.expression.replace('x^2', '**2').replace('sqrt', 'sqrt')
                
                result = self.safe_eval(eval_expr)
                
                if isinstance(result, float):
                    if result.is_integer():
                        result = int(result)
                    else:
                        result = round(result, 6) # تقريب النتائج العشرية الطويلة لمنع تجاوز الشاشة
                
                self.display_var.set(str(result))
                self.expression = str(result)
            except Exception:
                self.display_var.set("Error")
                self.expression = ""
        else:
            # التعامل مع الأزرار الخاصة والعمليات المتقدمة
            if char == "x^2":
                self.expression += "**2"
            elif char == "sqrt":
                self.expression += "sqrt("
            elif char in ["sin", "cos", "tan"]:
                self.expression += f"{char}("
            elif char == "pi":
                self.expression += "pi"
            elif char == "e":
                self.expression += "e"
            else:
                if char == ".":
                    parts = re.split(r'[\+\-\*\/\(\)\^\s]', self.expression)
                    current_number = parts[-1] if parts else ""
                    if "." in current_number:
                        return

                if len(self.expression) >= 150:
                    return

                if self.display_var.get() == "0" and char not in ".+-*/^":
                    self.expression = char
                else:
                    self.expression += str(char)
            
            self.display_var.set(self.expression)

if __name__ == "__main__":
    try:
        root = tk.Tk()
        app = AICalculatorPro(root)
        root.mainloop()
    except Exception as e:
        messagebox.showerror("Critical Error", f"An unexpected error occurred:\n{e}")
